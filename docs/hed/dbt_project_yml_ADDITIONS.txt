# COMPLETE EXAMPLE SHOWING BOTH CDS AND HED
# =============================================================================

models:
  dbt_industries_views:
    # CDS Healthcare Models
    cds:
      +tags: ['cds', 'healthcare']
      +schema: industries_healthcare
      +materialized: view
    
    # HED Higher Education Models (NEW)
    hed:
      +tags: ['hed', 'education']
      +schema: industries_education
      +materialized: view

vars:
  # Source configuration for CDS Healthcare
  cds_source_database: 'HOL_DATABASE'
  cds_source_schema: 'INDUSTRIES_HEALTHCARE'
  cds_source_table: 'CDS_RECORDS'
  
  # Source configuration for HED Higher Education (NEW)
  hed_source_database: 'HOL_DATABASE'
  hed_source_schema: 'INDUSTRIES_HIGHER_EDUCATION'
  hed_source_table: 'HED_RECORDS'


# =============================================================================
# WHAT THESE SETTINGS DO
# =============================================================================

# +tags: ['hed', 'education']
#   → Allows selective execution: dbt run --select tag:hed
#   → Enables filtering: dbt run --select tag:education

# +schema: industries_education
#   → Creates views in HOL_DATABASE.INDUSTRIES_EDUCATION schema
#   → Keeps HED views separate from CDS (healthcare) views

# +materialized: view
#   → Creates standard views (not tables or incremental)
#   → Query-on-demand, no additional storage cost
#   → Change to 'table' if you need better query performance

# hed_source_database: 'HOL_DATABASE'
# hed_source_schema: 'INDUSTRIES_HIGHER_EDUCATION'  
# hed_source_table: 'HED_RECORDS'
#   → These variables are used in _hed__sources.yml
#   → Allows easy environment-specific configuration
#   → Can override per environment: --vars "hed_source_schema: DEV_SCHEMA"
